OBJECTIVE
---------
Design and implement the foundational database schema supporting the entire
clinic management platform with multi-tenant isolation, HIPAA compliance,
and support for all 28+ modules.

BUSINESS LOGIC
--------------
Database Architecture Principles:

1. UNIVERSAL DATA MODEL
   Every entity follows consistent pattern:
   - id: UUID (primary key)
   - tenant_id: UUID (multi-tenant isolation)
   - created_at: TIMESTAMP (audit trail)
   - updated_at: TIMESTAMP (change tracking)
   - created_by: UUID (user who created)
   - updated_by: UUID (user who updated)
   - is_deleted: BOOLEAN (soft delete for HIPAA)

2. REFERENTIAL INTEGRITY
   - Foreign keys on all relationships
   - CASCADE deletes only for non-sensitive data
   - Soft deletes for patient/medical data
   - ON DELETE RESTRICT for critical entities

3. MULTI-TENANCY PATTERN
   - tenant_id on every table
   - Composite primary keys include tenant_id
   - Partitioning by tenant_id for large tables
   - Separate indexes per tenant for query performance

4. CLINICAL DATA PRINCIPLES
   - Immutable medical records (no updates after creation)
   - Audit trail on all changes
   - Soft delete for regulatory compliance
   - Version tracking for medical changes

CORE ENTITIES STRUCTURE
-----------------------
Platform Level:
├── tenants (SaaS clinics)
├── subscription_plans
├── billing_accounts
└── audit_logs (cross-tenant)

Clinic Level (all scoped to tenant_id):
├── clinic_config
├── users (staff + admins)
├── patients (core clinical data)
├── medical_records (EMR structure)
├── appointments (scheduling)
├── consultations (clinical encounters)
├── prescriptions (medication management)
├── diagnoses
├── lab_orders
├── lab_results
├── invoices (billing)
├── payments
├── inventory (pharmacy/supplies)
├── document_store (files/attachments)
├── notifications
└── system_logs (clinic-specific)

DATABASE SCHEMA - PART 1: PLATFORM CORE
---------------------------------------

-- Tenant Management
CREATE TABLE tenants (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(255) NOT NULL,
  subdomain VARCHAR(100) UNIQUE NOT NULL,
  status ENUM('active', 'suspended', 'trial_ended') DEFAULT 'active',
  subscription_plan_id UUID NOT NULL,
  
  -- Clinic Information
  phone VARCHAR(20),
  email VARCHAR(255),
  address TEXT,
  city VARCHAR(100),
  state VARCHAR(50),
  postal_code VARCHAR(20),
  country VARCHAR(100),
  timezone VARCHAR(50),
  
  -- Configuration
  primary_currency VARCHAR(3) DEFAULT 'USD',
  enable_telemedicine BOOLEAN DEFAULT TRUE,
  enable_offline_mode BOOLEAN DEFAULT TRUE,
  max_concurrent_users INT DEFAULT 100,
  
  -- Billing
  monthly_fee DECIMAL(10,2),
  billing_cycle_start_date DATE,
  
  -- Dates
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW(),
  trial_end_date DATE,
  
  INDEX idx_subdomain (subdomain),
  INDEX idx_status (status)
);

CREATE TABLE subscription_plans (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(100) NOT NULL,
  description TEXT,
  price DECIMAL(10,2),
  max_users INT,
  max_patients INT,
  features JSONB, -- {"telemedicine": true, "api_access": false, ...}
  billing_interval ENUM('monthly', 'annual') DEFAULT 'monthly',
  created_at TIMESTAMP DEFAULT NOW()
);

DATABASE SCHEMA - PART 2: USERS & ACCESS CONTROL
-------------------------------------------------

CREATE TABLE users (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id UUID NOT NULL REFERENCES tenants(id),
  
  -- Authentication
  email VARCHAR(255) NOT NULL,
  password_hash VARCHAR(255) NOT NULL,
  
  -- Profile
  first_name VARCHAR(100),
  last_name VARCHAR(100),
  phone VARCHAR(20),
  avatar_url VARCHAR(500),
  
  -- Account Status
  status ENUM('active', 'inactive', 'suspended') DEFAULT 'active',
  is_email_verified BOOLEAN DEFAULT FALSE,
  is_mfa_enabled BOOLEAN DEFAULT FALSE,
  mfa_method VARCHAR(20), -- 'sms', 'totp', null
  
  -- Audit
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW(),
  created_by UUID,
  updated_by UUID,
  is_deleted BOOLEAN DEFAULT FALSE,
  
  -- Access
  last_login TIMESTAMP,
  password_changed_at TIMESTAMP,
  
  UNIQUE(tenant_id, email),
  INDEX idx_tenant_email (tenant_id, email),
  INDEX idx_status (status)
);

CREATE TABLE roles (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id UUID NOT NULL REFERENCES tenants(id),
  name VARCHAR(100) NOT NULL,
  description TEXT,
  created_at TIMESTAMP DEFAULT NOW(),
  
  UNIQUE(tenant_id, name)
);

CREATE TABLE permissions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  resource VARCHAR(100) NOT NULL, -- 'patient', 'prescription', etc.
  action VARCHAR(50) NOT NULL, -- 'read', 'write', 'delete', 'approve'
  description TEXT,
  created_at TIMESTAMP DEFAULT NOW(),
  
  UNIQUE(resource, action)
);

CREATE TABLE role_permissions (
  role_id UUID NOT NULL REFERENCES roles(id) ON DELETE CASCADE,
  permission_id UUID NOT NULL REFERENCES permissions(id) ON DELETE CASCADE,
  tenant_id UUID NOT NULL,
  created_at TIMESTAMP DEFAULT NOW(),
  
  PRIMARY KEY (role_id, permission_id)
);

CREATE TABLE user_roles (
  user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  role_id UUID NOT NULL REFERENCES roles(id) ON DELETE CASCADE,
  tenant_id UUID NOT NULL,
  assigned_at TIMESTAMP DEFAULT NOW(),
  
  PRIMARY KEY (user_id, role_id)
);

DATABASE SCHEMA - PART 3: PATIENT & MEDICAL DATA
------------------------------------------------

CREATE TABLE patients (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id UUID NOT NULL REFERENCES tenants(id),
  
  -- Demographics
  first_name VARCHAR(100) NOT NULL,
  last_name VARCHAR(100) NOT NULL,
  date_of_birth DATE,
  gender ENUM('M', 'F', 'Other', 'Prefer not to say'),
  phone VARCHAR(20),
  email VARCHAR(255),
  
  -- Address
  address TEXT,
  city VARCHAR(100),
  state VARCHAR(50),
  postal_code VARCHAR(20),
  country VARCHAR(100),
  
  -- Medical Identifiers
  medical_record_number VARCHAR(50), -- Clinic-specific MRN
  national_id VARCHAR(50), -- SSN, etc (encrypted)
  insurance_id VARCHAR(50),
  insurance_group VARCHAR(50),
  insurance_provider VARCHAR(255),
  
  -- Relationships
  primary_doctor_id UUID REFERENCES users(id),
  referred_by VARCHAR(255),
  
  -- Status
  status ENUM('active', 'inactive', 'archived') DEFAULT 'active',
  
  -- Clinical
  blood_type VARCHAR(3),
  allergies TEXT,
  chronic_conditions TEXT, -- Denormalized for quick access
  
  -- Consent & Privacy
  has_consented_to_treatment BOOLEAN DEFAULT FALSE,
  consent_date DATE,
  privacy_policy_accepted BOOLEAN DEFAULT FALSE,
  
  -- Dates
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW(),
  created_by UUID NOT NULL,
  updated_by UUID NOT NULL,
  is_deleted BOOLEAN DEFAULT FALSE,
  
  INDEX idx_tenant_patient (tenant_id, medical_record_number),
  INDEX idx_status (status),
  INDEX idx_primary_doctor (primary_doctor_id)
);

CREATE TABLE medical_records (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id UUID NOT NULL REFERENCES tenants(id),
  patient_id UUID NOT NULL REFERENCES patients(id),
  
  -- Record Information
  visit_date DATE NOT NULL,
  encounter_type ENUM('clinic_visit', 'telemedicine', 'follow_up', 'emergency'),
  chief_complaint TEXT,
  history_of_present_illness TEXT,
  
  -- Physical Examination
  vital_signs JSONB, -- {temp, bp, pulse, respiration, weight, height}
  physical_exam_notes TEXT,
  
  -- Assessment & Plan
  assessment TEXT,
  diagnosis_ids UUID[], -- References to diagnoses
  treatment_plan TEXT,
  
  -- Clinical Staff
  examined_by UUID NOT NULL REFERENCES users(id),
  recorded_by UUID NOT NULL,
  
  -- Status
  status ENUM('draft', 'completed', 'signed') DEFAULT 'draft',
  is_locked BOOLEAN DEFAULT FALSE, -- Prevent editing after signing
  
  -- Dates
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW(),
  created_by UUID NOT NULL,
  updated_by UUID NOT NULL,
  is_deleted BOOLEAN DEFAULT FALSE,
  
  INDEX idx_tenant_patient (tenant_id, patient_id),
  INDEX idx_visit_date (visit_date),
  INDEX idx_status (status)
);

CREATE TABLE diagnoses (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id UUID NOT NULL REFERENCES tenants(id),
  patient_id UUID NOT NULL REFERENCES patients(id),
  
  -- Diagnosis Information
  icd10_code VARCHAR(10) NOT NULL, -- International Classification of Diseases
  description VARCHAR(500),
  severity ENUM('mild', 'moderate', 'severe', 'critical'),
  
  -- Timeline
  diagnosis_date DATE NOT NULL,
  resolved_date DATE, -- NULL if ongoing
  
  -- Related
  medical_record_id UUID REFERENCES medical_records(id),
  
  -- Audit
  created_at TIMESTAMP DEFAULT NOW(),
  created_by UUID NOT NULL,
  is_deleted BOOLEAN DEFAULT FALSE,
  
  INDEX idx_tenant_patient (tenant_id, patient_id),
  INDEX idx_icd10 (icd10_code)
);

DATABASE SCHEMA - PART 4: APPOINTMENTS & CONSULTATIONS
-----------------------------------------------------

CREATE TABLE appointments (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id UUID NOT NULL REFERENCES tenants(id),
  
  -- Participants
  patient_id UUID NOT NULL REFERENCES patients(id),
  doctor_id UUID NOT NULL REFERENCES users(id),
  
  -- Scheduling
  scheduled_date DATE NOT NULL,
  scheduled_time TIME NOT NULL,
  duration_minutes INT DEFAULT 30,
  
  -- Type
  appointment_type VARCHAR(50), -- 'consultation', 'follow-up', 'emergency'
  reason_for_visit TEXT,
  
  -- Status
  status ENUM('scheduled', 'checked_in', 'completed', 'cancelled', 'no_show') DEFAULT 'scheduled',
  cancellation_reason VARCHAR(500),
  cancelled_by UUID REFERENCES users(id),
  
  -- Telemedicine
  is_telemedicine BOOLEAN DEFAULT FALSE,
  meeting_url VARCHAR(500), -- Zoom, Teams, etc.
  
  -- Notifications
  reminder_sent_at TIMESTAMP,
  
  -- Dates
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW(),
  created_by UUID NOT NULL,
  updated_by UUID NOT NULL,
  is_deleted BOOLEAN DEFAULT FALSE,
  
  INDEX idx_tenant_date (tenant_id, scheduled_date),
  INDEX idx_patient (patient_id),
  INDEX idx_doctor (doctor_id),
  INDEX idx_status (status)
);

CREATE TABLE consultations (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id UUID NOT NULL REFERENCES tenants(id),
  appointment_id UUID REFERENCES appointments(id),
  
  -- Participants
  patient_id UUID NOT NULL REFERENCES patients(id),
  doctor_id UUID NOT NULL REFERENCES users(id),
  
  -- Notes
  subjective_findings TEXT, -- What patient reports
  objective_findings TEXT, -- What doctor observes
  assessment TEXT, -- Doctor's assessment
  plan TEXT, -- Treatment plan
  
  -- Outcomes
  status ENUM('in_progress', 'completed', 'pending_review') DEFAULT 'in_progress',
  
  -- Dates
  started_at TIMESTAMP NOT NULL DEFAULT NOW(),
  completed_at TIMESTAMP,
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW(),
  created_by UUID NOT NULL,
  updated_by UUID NOT NULL,
  is_deleted BOOLEAN DEFAULT FALSE,
  
  INDEX idx_tenant_patient (tenant_id, patient_id),
  INDEX idx_doctor (doctor_id)
);

DATABASE SCHEMA - PART 5: PRESCRIPTIONS & MEDICATIONS
----------------------------------------------------

CREATE TABLE prescriptions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id UUID NOT NULL REFERENCES tenants(id),
  
  -- Who & Why
  patient_id UUID NOT NULL REFERENCES patients(id),
  doctor_id UUID NOT NULL REFERENCES users(id),
  consultation_id UUID REFERENCES consultations(id),
  
  -- Medication
  drug_name VARCHAR(255) NOT NULL,
  drug_code VARCHAR(50), -- NDC code, etc.
  strength VARCHAR(50), -- e.g., "500mg"
  dosage_form VARCHAR(50), -- tablet, liquid, injection
  
  -- Dosage Instructions
  quantity INT NOT NULL,
  unit ENUM('tablet', 'ml', 'vial', 'bottle'),
  frequency VARCHAR(100), -- "3 times daily", "every 8 hours"
  duration_days INT, -- How long to take
  route ENUM('oral', 'injection', 'topical', 'inhalation'),
  special_instructions TEXT, -- "with food", "avoid dairy"
  
  -- Authorization
  is_approved BOOLEAN DEFAULT FALSE,
  approved_by UUID REFERENCES users(id),
  approved_at TIMESTAMP,
  
  -- Fulfillment
  is_filled BOOLEAN DEFAULT FALSE,
  filled_at TIMESTAMP,
  filled_by UUID REFERENCES users(id),
  pharmacy_name VARCHAR(255),
  
  -- Status
  status ENUM('pending', 'approved', 'filled', 'expired', 'cancelled') DEFAULT 'pending',
  expiry_date DATE, -- When prescription expires (usually 1 year)
  refills_allowed INT DEFAULT 0,
  refills_used INT DEFAULT 0,
  
  -- Dates
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW(),
  created_by UUID NOT NULL,
  updated_by UUID NOT NULL,
  is_deleted BOOLEAN DEFAULT FALSE,
  
  INDEX idx_tenant_patient (tenant_id, patient_id),
  INDEX idx_status (status),
  INDEX idx_expiry (expiry_date)
);

DATABASE SCHEMA - PART 6: BILLING & PAYMENTS
--------------------------------------------

CREATE TABLE invoices (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id UUID NOT NULL REFERENCES tenants(id),
  
  -- Who
  patient_id UUID NOT NULL REFERENCES patients(id),
  
  -- Invoice Details
  invoice_number VARCHAR(50) NOT NULL,
  issue_date DATE NOT NULL,
  due_date DATE NOT NULL,
  
  -- Amounts
  subtotal DECIMAL(10,2) NOT NULL,
  tax_amount DECIMAL(10,2) DEFAULT 0,
  discount_amount DECIMAL(10,2) DEFAULT 0,
  total_amount DECIMAL(10,2) NOT NULL,
  
  -- Status
  status ENUM('draft', 'sent', 'partially_paid', 'paid', 'overdue', 'cancelled') DEFAULT 'draft',
  
  -- Line Items (in separate table)
  -- see invoice_line_items
  
  -- Notes
  notes TEXT,
  terms TEXT,
  
  -- Dates
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW(),
  created_by UUID NOT NULL,
  updated_by UUID NOT NULL,
  is_deleted BOOLEAN DEFAULT FALSE,
  
  UNIQUE(tenant_id, invoice_number),
  INDEX idx_patient (patient_id),
  INDEX idx_status (status),
  INDEX idx_due_date (due_date)
);

CREATE TABLE invoice_line_items (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  invoice_id UUID NOT NULL REFERENCES invoices(id) ON DELETE CASCADE,
  
  -- Item
  description VARCHAR(500),
  service_date DATE,
  quantity INT DEFAULT 1,
  unit_price DECIMAL(10,2),
  total_price DECIMAL(10,2),
  
  -- Category
  category ENUM('consultation', 'procedure', 'medication', 'lab', 'other')
);

CREATE TABLE payments (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id UUID NOT NULL REFERENCES tenants(id),
  
  -- What
  invoice_id UUID NOT NULL REFERENCES invoices(id),
  patient_id UUID NOT NULL REFERENCES patients(id),
  
  -- Payment Info
  amount DECIMAL(10,2) NOT NULL,
  payment_method ENUM('cash', 'card', 'check', 'insurance', 'bank_transfer'),
  
  -- Transaction
  transaction_id VARCHAR(100), -- Payment processor ID
  status ENUM('pending', 'completed', 'failed', 'refunded') DEFAULT 'pending',
  
  -- Dates
  payment_date DATE NOT NULL,
  created_at TIMESTAMP DEFAULT NOW(),
  
  INDEX idx_invoice (invoice_id),
  INDEX idx_status (status)
);

DATABASE SCHEMA - PART 7: COMPLIANCE & AUDIT
--------------------------------------------

CREATE TABLE audit_logs (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id UUID NOT NULL,
  
  -- Action
  action_type VARCHAR(100), -- 'create', 'read', 'update', 'delete', 'login', 'export'
  resource_type VARCHAR(100), -- 'patient', 'prescription', 'invoice'
  resource_id UUID,
  
  -- Actor
  user_id UUID,
  user_email VARCHAR(255),
  
  -- Details
  ip_address VARCHAR(45),
  user_agent TEXT,
  changes JSONB, -- What changed (old value, new value)
  status ENUM('success', 'failure'),
  error_message TEXT,
  
  -- Dates
  created_at TIMESTAMP DEFAULT NOW(),
  
  INDEX idx_tenant (tenant_id),
  INDEX idx_resource (resource_type, resource_id),
  INDEX idx_user (user_id),
  INDEX idx_action_type (action_type),
  INDEX idx_created_at (created_at)
);

CREATE TABLE consent_logs (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id UUID NOT NULL,
  patient_id UUID NOT NULL REFERENCES patients(id),
  
  -- Consent
  consent_type VARCHAR(100), -- 'treatment', 'privacy', 'telemedicine'
  is_given BOOLEAN,
  
  -- Details
  document_version VARCHAR(20),
  ip_address VARCHAR(45),
  device_info VARCHAR(500),
  
  -- Signature (if applicable)
  signature_image_url VARCHAR(500),
  
  -- Dates
  created_at TIMESTAMP DEFAULT NOW(),
  
  INDEX idx_patient (patient_id),
  INDEX idx_type (consent_type)
);

CRITICAL INDEXES FOR PERFORMANCE
---------------------------------
-- Multi-tenant queries (used on almost every query)
CREATE INDEX idx_tenant_status ON appointments(tenant_id, status);
CREATE INDEX idx_tenant_date ON appointments(tenant_id, scheduled_date DESC);

-- Patient lookups
CREATE INDEX idx_patient_tenant ON patients(tenant_id, status);
CREATE INDEX idx_patient_mrn ON patients(tenant_id, medical_record_number);

-- Medical record queries
CREATE INDEX idx_medical_record_patient ON medical_records(tenant_id, patient_id, visit_date DESC);

-- Audit trail queries
CREATE INDEX idx_audit_time_range ON audit_logs(tenant_id, created_at DESC);

-- Billing
CREATE INDEX idx_invoice_due ON invoices(tenant_id, due_date, status);

CONSTRAINTS & RULES
------------------
1. Soft Deletes: Never physically delete patient/medical records
2. Immutability: Medical records cannot be edited after 24 hours (or mark as amended)
3. Cascade: Only cascade delete for non-clinical data
4. Uniqueness: medical_record_number unique per clinic
5. Foreign Keys: All relationships must have FK constraints
6. Timestamps: All records must have created_at, updated_at
7. Audit Trail: created_by, updated_by on all tables
8. Tenant Isolation: Every query MUST filter by tenant_id
